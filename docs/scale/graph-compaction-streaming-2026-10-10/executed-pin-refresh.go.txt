package sim

import (
	"path/filepath"
	"reflect"
	"testing"
)

func TestRefreshStreamingCompactionPinsDevelopment(t *testing.T) {
	paths, err := filepath.Glob("testdata/regressions/graph-compaction-*.json")
	if err != nil || len(paths) != 14 {
		t.Fatal(paths, err)
	}
	for _, path := range paths {
		old, err := LoadTrace(path)
		if err != nil {
			t.Fatal(err)
		}
		next, err := runGraphCompaction(old.Seed, nil)
		if err != nil {
			t.Fatal(path, err)
		}
		before, after := old, next
		before.Transport, after.Transport = streamingNonGets(old.Transport), streamingNonGets(next.Transport)
		if !reflect.DeepEqual(before, after) {
			t.Fatal("non-Get trace changed", path)
		}
		if err = next.Save(path); err != nil {
			t.Fatal(err)
		}
	}
}

func streamingNonGets(events []TransportEvent) []TransportEvent {
	out := []TransportEvent{}
	for _, event := range events {
		if event.Operation != "graph_publication_get" {
			out = append(out, event)
		}
	}
	return out
}
