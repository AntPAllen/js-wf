package sim
import (
 "crypto/sha256"
 "encoding/hex"
 "encoding/json"
 "os"
 "path/filepath"
 "reflect"
 "testing"
)
func TestMigrateDuplicateReaderObservations(t *testing.T) {
 if os.Getenv("MIGRATE_DUPLICATE_READ_ROOT")!="1" {t.Fatal("explicit migration required")}
 paths,e:=filepath.Glob("testdata/regressions/*.json");if e!=nil||len(paths)!=946{t.Fatal("inventory",e,len(paths))}
 type change struct {Path,BeforeSHA256,AfterSHA256 string;Removed int}
 changes:=[]change{}
 digest:=func(b []byte)string{h:=sha256.Sum256(b);return hex.EncodeToString(h[:])}
 for _,path:=range paths {
  raw,e:=os.ReadFile(path);if e!=nil{t.Fatal(e)}
  old,e:=LoadTrace(path);if e!=nil{t.Fatal(e)}
  guided:=old;guided.GuidanceMask=make([]bool,len(old.Decisions));for i:=range guided.GuidanceMask{guided.GuidanceMask[i]=true}
  fresh,e:=replayTrace(guided);if e!=nil{t.Fatalf("%s: %v",path,e)}
  a,b:=old,fresh;a.Transport=nil;b.Transport=nil;a.GuidanceMask=nil;b.GuidanceMask=nil
  if !reflect.DeepEqual(a,b){t.Fatalf("metadata/decisions changed: %s",path)}
  j,removed:=0,0
  for i,event:=range old.Transport {
   if j<len(fresh.Transport)&&reflect.DeepEqual(event,fresh.Transport[j]) {j++;continue}
   if event.Operation!="read_root" || i==0 || !reflect.DeepEqual(event,old.Transport[i-1]) {t.Fatalf("nonduplicate observation changed: %s at %d (new %d)",path,i,j)}
   removed++
  }
  if j!=len(fresh.Transport){t.Fatalf("transport appended: %s",path)}
  exact,e:=replayTrace(fresh);if e!=nil||!reflect.DeepEqual(fresh,exact){t.Fatalf("new exact replay failed: %s %v",path,e)}
  if removed!=0 {
   if e=fresh.Save(path);e!=nil{t.Fatal(e)};after,e:=os.ReadFile(path);if e!=nil{t.Fatal(e)}
   changes=append(changes,change{path,digest(raw),digest(after),removed})
  }
 }
 raw,_:=json.MarshalIndent(changes,"","  ");if e=os.WriteFile(os.Getenv("MIGRATE_REPORT"),append(raw,'\n'),0600);e!=nil{t.Fatal(e)}
 t.Logf("verified 946 unchanged seed/decision/invariant controls; migrated %d files by duplicate-read deletion only",len(changes))
}
