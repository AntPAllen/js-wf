package sim
import("testing";"path/filepath";"reflect")
func TestRefreshCompactionPinsDevelopment(t *testing.T) {
 paths,err:=filepath.Glob("testdata/regressions/graph-compaction-*.json");if err!=nil || len(paths)!=14 {t.Fatal(paths,err)}
 for _,path:=range paths { old,err:=LoadTrace(path);if err!=nil {t.Fatal(err)};next,err:=runGraphCompaction(old.Seed,nil);if err!=nil {t.Fatal(path,err)}
 if !reflect.DeepEqual(old.Decisions,next.Decisions) || !reflect.DeepEqual(compactionNonReads(old.Transport),compactionNonReads(next.Transport)) {t.Fatal("semantic decisions/events changed",path)}
 if err=next.Save(path);err!=nil {t.Fatal(err)}
 }
}

func compactionNonReads(events []TransportEvent) []TransportEvent { out:=[]TransportEvent{};for _,e:=range events {if e.Operation!="graph_publication_get" {out=append(out,e)}};return out}
