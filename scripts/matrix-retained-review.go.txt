// Standalone review helper: build from a copied .go file in this module.
// The caller supplies ONLY a fresh copy of closed R3 testcluster stores and
// retains executable/source identity and original/copy hashes externally.
package main

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"github.com/anishathalye/porcupine"
	"js-wf/client"
	"js-wf/history"
	"os"
	"path/filepath"
	"strconv"
	"time"

	"github.com/nats-io/nats.go/jetstream"
	"js-wf/integrity"
	"js-wf/provision"
	"js-wf/testcluster"
)

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
func run() error {
	if len(os.Args) != 6 || !filepath.IsAbs(os.Args[1]) {
		return fmt.Errorf("usage: matrix-retained-review ABSOLUTE_COPIED_CLUSTER_ROOT COUNT ENTRIES HISTORY_JSONL OUTPUT_JSON")
	}
	count, err := strconv.Atoi(os.Args[2])
	if err != nil || count < 1 {
		return fmt.Errorf("invalid expected count")
	}
	entries, err := strconv.Atoi(os.Args[3])
	if err != nil || entries < count {
		return fmt.Errorf("invalid expected entries")
	}
	file, err := os.Open(os.Args[4])
	if err != nil {
		return err
	}
	var operations []client.Operation
	scanner := bufio.NewScanner(file)
	scanner.Buffer(make([]byte, 64*1024), 2*1024*1024)
	for scanner.Scan() {
		var op client.Operation
		if err := json.Unmarshal(scanner.Bytes(), &op); err != nil {
			file.Close()
			return err
		}
		operations = append(operations, op)
	}
	scanErr := scanner.Err()
	file.Close()
	if scanErr != nil {
		return scanErr
	}
	if len(operations) == 0 {
		return fmt.Errorf("empty history")
	}
	for _, check := range []func([]client.Operation, time.Duration) (porcupine.CheckResult, error){history.CheckStarts, history.CheckSignals, history.CheckResults} {
		result, err := check(operations, 30*time.Second)
		if err != nil || result != porcupine.Ok {
			return fmt.Errorf("copied history=%v: %w", result, err)
		}
	}
	root := os.Args[1]
	// External observer gets time to capture /proc executable identity before
	// opening copies. This delay is outside admission/audit budgets.
	time.Sleep(250 * time.Millisecond)
	cluster, err := testcluster.StartProcesses(root, 3)
	if err != nil {
		return err
	}
	defer cluster.Close()
	all := make([]jetstream.JetStream, 3)
	for i, nc := range cluster.Clients {
		all[i], err = jetstream.New(nc)
		if err != nil {
			return err
		}
	}
	// Readiness observes existing assets; it never provisions or alters them.
	ready, stopReady := context.WithTimeout(context.Background(), 15*time.Second)
	var roles = []string{"WF_INV", "WF_JRN", "WF_RUN", "KV_WF_STATE"}
	for _, name := range roles {
		admitted := false
		for ready.Err() == nil {
			call, stop := context.WithTimeout(ready, 2*time.Second)
			stream, readErr := all[0].Stream(call, name)
			var info *jetstream.StreamInfo
			if readErr == nil {
				info, readErr = stream.Info(call)
			}
			stop()
			if readErr == nil && info.Cluster != nil && info.Cluster.Leader != "" && len(info.Cluster.Replicas) == 2 {
				current := true
				for _, peer := range info.Cluster.Replicas {
					current = current && peer.Current && !peer.Offline
				}
				if current {
					admitted = true
					break
				}
			}
			time.Sleep(20 * time.Millisecond)
		}
		if !admitted {
			stopReady()
			return fmt.Errorf("copied source %s not current: %w", name, ready.Err())
		}
	}
	stopReady()
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	began := time.Now()
	report, err := integrity.CheckWithStreamingStateReads(ctx, all[0])
	if err != nil {
		return fmt.Errorf("full copied retained audit: %w", err)
	}
	elapsed := time.Since(began)
	if report.Invocations != count || report.Journals != count || report.Entries != entries || report.Terminal != count {
		return fmt.Errorf("full copied retained counts %+v expected=%d", report, count)
	}
	// Stream.Info is leader-routed even through pinned clients. Retain these
	// API views and separately read each server's local monitoring state below.
	var queue = []*jetstream.StreamInfo{}
	for i, js := range all {
		stream, err := js.Stream(ctx, "WF_RUN")
		if err != nil {
			return err
		}
		info, err := stream.Info(ctx)
		if err != nil {
			return err
		}
		if info.State.Msgs != 0 || info.State.Consumers != int(provision.Partitions) {
			return fmt.Errorf("peer%d queue not drained: %+v", i, info.State)
		}
		queue = append(queue, info)
	}
	run, err := all[0].Stream(ctx, "WF_RUN")
	if err != nil {
		return err
	}
	consumers := run.ListConsumers(ctx)
	checked := 0
	for info := range consumers.Info() {
		if info.NumPending != 0 || info.NumAckPending != 0 {
			return fmt.Errorf("durable %s not drained pending=%d ack=%d", info.Name, info.NumPending, info.NumAckPending)
		}
		checked++
	}
	if err := consumers.Err(); err != nil {
		return err
	}
	if checked != int(provision.Partitions) {
		return fmt.Errorf("incomplete durable census %d expected=%d", checked, provision.Partitions)
	}
	// These sequential pinned monitoring reads share the original 20-second
	// audit budget. They observe local stores, but are not simultaneous.
	type physicalPeer struct {
		Node     int             `json:"node"`
		Started  time.Time       `json:"started"`
		Finished time.Time       `json:"finished"`
		State    json.RawMessage `json:"state"`
	}
	physical := make([]physicalPeer, 3)
	identities := make(map[string]bool)
	for node := range physical {
		p := &physical[node]
		p.Node, p.Started = node, time.Now().UTC()
		data, err := cluster.Diagnostic(ctx, node, "jetstream")
		p.Finished = time.Now().UTC()
		if err != nil {
			return fmt.Errorf("peer%d local monitoring: %w", node, err)
		}
		var snapshot struct {
			ServerID string `json:"server_id"`
			Accounts []struct {
				Streams []struct {
					Name  string `json:"name"`
					State struct {
						Messages  *uint64 `json:"messages"`
						Consumers *int    `json:"consumer_count"`
					} `json:"state"`
				} `json:"stream_detail"`
			} `json:"account_details"`
		}
		if err := json.Unmarshal(data, &snapshot); err != nil {
			return err
		}
		if snapshot.ServerID == "" || identities[snapshot.ServerID] {
			return fmt.Errorf("missing or duplicate physical server identity")
		}
		identities[snapshot.ServerID] = true
		found := 0
		for _, account := range snapshot.Accounts {
			for _, stream := range account.Streams {
				if stream.Name != "WF_RUN" {
					continue
				}
				found++
				if stream.State.Messages == nil || *stream.State.Messages != 0 || stream.State.Consumers == nil || *stream.State.Consumers != int(provision.Partitions) {
					return fmt.Errorf("peer%d local physical queue not drained: %+v", node, stream.State)
				}
			}
		}
		if found != 1 {
			return fmt.Errorf("peer%d local WF_RUN census=%d", node, found)
		}
		p.State = data
	}
	if ctx.Err() != nil {
		return ctx.Err()
	}
	result := struct {
		Scope         string                  `json:"scope"`
		Report        integrity.Report        `json:"report"`
		AuditNS       int64                   `json:"audit_ns"`
		WholeReviewNS int64                   `json:"whole_review_ns"`
		Queue         []*jetstream.StreamInfo `json:"queue_all_three_peers"`
		Physical      []physicalPeer          `json:"physical_queue_peers"`
		Consumers     int                     `json:"drained_partition_consumers"`
	}{"complete copied process-matrix retained integrity/drain and independently rechecked history; not original fault verdict or full-matrix qualification", report, elapsed.Nanoseconds(), time.Since(began).Nanoseconds(), queue, physical, checked}
	data, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		return err
	}
	if err := os.WriteFile(os.Args[5], append(data, '\n'), 0600); err != nil {
		return err
	}
	fmt.Printf("full retained audit invocations=%d journals=%d entries=%d terminal=%d audit=%s review=%s drained_consumers=%d\n", report.Invocations, report.Journals, report.Entries, report.Terminal, elapsed, time.Since(began), checked)
	return nil
}
